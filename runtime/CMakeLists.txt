cmake_minimum_required(VERSION 3.18)
project(PvZFEnglish LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 17)
set(DOBBY_DEBUG OFF CACHE BOOL "" FORCE)
set(DOBBY_BUILD_EXAMPLE OFF CACHE BOOL "" FORCE)
set(DOBBY_BUILD_TEST OFF CACHE BOOL "" FORCE)
set(Plugin.SymbolResolver OFF CACHE BOOL "" FORCE)
add_subdirectory(${DOBBY_SOURCE} dobby-build)
add_library(pvzfenglish SHARED english.cpp)
option(PVZF_MUSIC_DIAGNOSTICS "Enumerate scene audio sources after music changes" OFF)
if(PVZF_MUSIC_DIAGNOSTICS)
    target_compile_definitions(pvzfenglish PRIVATE PVZF_MUSIC_DIAGNOSTICS=1)
endif()
target_include_directories(pvzfenglish PRIVATE ${DOBBY_SOURCE}/include)
target_link_libraries(pvzfenglish dobby_static log android dl)
target_link_options(pvzfenglish PRIVATE "-Wl,-z,max-page-size=16384")
